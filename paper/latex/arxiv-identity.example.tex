% Template for arxiv-identity.tex: the de-anonymized author block of the arXiv version.

% Labels (aff1, aff2, ...) link authors to affiliations; an author may carry several, as {aff1,aff2}.

  \mlsysaffiliation{aff1}{Department, University, City, Country}
  \mlsysaffiliation{aff2}{Department, University, City, Country}}
\newcommand{\ArxivCorrespondence}{\mlsyscorrespondingauthor{First Author}{first.author@example.org}}
\newcommand{\ArxivPdfAuthor}{First Author and Second Author}
